 import json
import boto3
import uuid

dynamodb = boto3.resource("dynamodb")
table = dynamodb.Table("StudentRegistration")


def lambda_handler(event, context):
    try:
        # Get registration data from the form
        if "body" in event and event["body"]:
            body = event["body"]

            if isinstance(body, str):
                data = json.loads(body)
            else:
                data = body
        else:
            data = event

        # Create a unique registration ID
        registration_id = str(uuid.uuid4())

        # Store the data in DynamoDB
        item = {
            "registration_id": registration_id,
            "name": data["name"],
            "email": data["email"],
            "phone": data["phone"]
        }

        table.put_item(Item=item)

        return {
            "statusCode": 200,
            "headers": {
                "Content-Type": "application/json"
            },
            "body": json.dumps({
                "message": "Registration successful",
                "registration_id": registration_id
            })
        }

    except (KeyError, json.JSONDecodeError) as e:
        return {
            "statusCode": 400,
            "body": json.dumps({
                "message": "Missing or invalid registration data",
                "error": str(e)
            })
        }

    except Exception:
        print("Error saving registration data")
        raise

import pendulum

from airflow.sdk import dag, task


@dag(
    dag_id="wednesday_datetime_dag",
    schedule="30 9 * * 3",
    start_date=pendulum.datetime(
        2026, 1, 1, tz="Asia/Kolkata"
    ),
    catchup=False,
    tags=["practical", "datetime"],
)
def wednesday_datetime_dag():

    @task
    def display_datetime():
        current_time = pendulum.now("Asia/Kolkata")

        print(
            "Current Date:",
            current_time.strftime("%Y-%m-%d")
        )
        print(
            "Current Time:",
            current_time.strftime("%H:%M:%S")
        )
        print("Time Zone: Asia/Kolkata")

    display_datetime()
